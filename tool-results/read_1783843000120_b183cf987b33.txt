    46→/** Derive a printable display name from a Member record. */
    47→function memberDisplayName(member: Member | null): string {
    48→  if (!member) return "—";
    49→  const isCompany =
    50→    member.membershipType === "COMPANY" ||
    51→    member.membershipType === "NPO_NGO" ||
    52→    member.membershipType === "SOLE_PROPRIETOR";
    53→  if (isCompany && member.companyName) return member.companyName;
    54→  const parts = [member.firstName, member.lastName].filter(Boolean);
    55→  if (parts.length) return parts.join(" ");
    56→  return member.email;
    57→}

   249→export function SharesView() {
   250→  const { currentMember } = useKasiStore();
   251→  const [data, setData] = useState<SharesData | null>(null);
   252→  const [loading, setLoading] = useState(true);
   253→  const [buyOpen, setBuyOpen] = useState(false);
   254→  const [selectedPhase, setSelectedPhase] = useState<number>(1);
   255→  const [quantity, setQuantity] = useState(10);
   256→  const [buying, setBuying] = useState(false);
   257→
   258→  async function load() {
   259→    if (!currentMember) return;
   260→    try {
   261→      const res = await fetch(`/api/shares?memberId=${currentMember.id}`, { cache: "no-store" });
   262→      if (res.ok) setData(await res.json());
   263→    } finally {
   264→      setLoading(false);
   265→    }
   266→  }
   267→
   268→  useEffect(() => {
   269→    load();
   270→  }, [currentMember]);
   271→
   272→  async function handleBuy() {
   273→    if (!currentMember) return;
   274→    setBuying(true);
   275→    try {
   276→      const res = await fetch("/api/shares/buy", {
   277→        method: "POST",
   278→        headers: { "Content-Type": "application/json" },
   279→        body: JSON.stringify({ memberId: currentMember.id, phase: selectedPhase, quantity }),
   280→      });
   281→      const result = await res.json();
   282→      if (!res.ok) {
   283→        toast.error(result.error || "Purchase failed");
   284→      } else {
   285→        toast.success(`Purchased ${quantity} share${quantity > 1 ? "s" : ""}! Certificate ${result.certificateNo} issued.`);
   286→        setBuyOpen(false);
   287→        await load();
   288→      }
   289→    } catch {
   290→      toast.error("Network error");
   291→    } finally {
   292→      setBuying(false);
