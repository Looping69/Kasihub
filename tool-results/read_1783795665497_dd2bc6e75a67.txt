    81→    81→export function RegistrationWizard() {
    82→    82→  const { closeRegistration, login } = useKasiStore();
    83→    83→  const [step, setStep] = useState<Step>("type");
    84→    84→  const [data, setData] = useState<FormData>(INITIAL);
    85→    85→  const [submitting, setSubmitting] = useState(false);
    86→    86→  const [newProfileNumber, setNewProfileNumber] = useState<string | null>(null);
    87→    87→
    88→    88→  const stepIndex = STEPS.findIndex((s) => s.key === step);
    89→    89→
    90→    90→  function update<K extends keyof FormData>(key: K, value: FormData[K]) {
    91→    91→    setData((d) => ({ ...d, [key]: value }));
    92→    92→  }
    93→    93→
    94→    94→  function next() {
    95→    95→    const order: Step[] = ["type", "details", "subscription", "kyc", "profile", "done"];
    96→    96→    const idx = order.indexOf(step);
    97→    97→    if (idx < order.length - 1) setStep(order[idx + 1]);
    98→    98→  }
    99→    99→  function prev() {
   100→   100→    const order: Step[] = ["type", "details", "subscription", "kyc", "profile"];
   101→   101→    const idx = order.indexOf(step);
   102→   102→    if (idx > 0) setStep(order[idx - 1]);
   103→   103→  }
   104→   104→
   105→   105→  const [newMember, setNewMember] = useState<{ id: string; profileNumber: string; firstName: string | null; lastName: string | null; companyName: string | null; email: string; membershipType: string } | null>(null);
   106→   106→
   107→   107→  async function submit() {
   108→   108→    setSubmitting(true);
   109→   109→    try {
   110→   110→      const res = await fetch("/api/members", {
   111→   111→        method: "POST",
   112→   112→        headers: { "Content-Type": "application/json" },
   113→   113→        body: JSON.stringify(data),
   114→   114→      });
   115→   115→      const result = await res.json();
   116→   116→      if (!res.ok) {
   117→   117→        toast.error(result.error || "Registration failed");
   118→   118→        setSubmitting(false);
   119→   119→        return;
   120→   120→      }
   121→   121→      setNewProfileNumber(result.profileNumber);
   122→   122→      setNewMember(result.member);
   123→   123→      setStep("done");
   124→   124→      toast.success("Welcome to KaSiHUB!");
   125→   125→    } catch {
   126→   126→      toast.error("Network error. Please try again.");
   127→   127→      setSubmitting(false);
   128→   128→    }
   129→   129→  }
   130→   130→
   131→   131→  function finish() {
   132→   132→    if (newMember) {
   133→   133→      // Log in as the newly registered member (re-fetch full record for all fields)
   134→   134→      fetch(`/api/members?memberId=${newMember.id}`, { cache: "no-store" })
   135→   135→        .then((r) => r.json())
   136→   136→        .then((d) => {
   137→   137→          if (d.member) {
   138→   138→            login(d.member.id, d.member);
   139→   139→          }
   140→   140→        })
   141→   141→        .catch(() => {})
   142→   142→        .finally(() => {
   143→   143→          closeRegistration();
   144→   144→        });
   145→   145→    } else {
   146→   146→      closeRegistration();
   147→   147→    }
   148→   148→  }
   149→   149→
   150→   150→  return (
   151→   151→    <Dialog open onOpenChange={(o) => !o && !submitting && closeRegistration()}>
   152→   152→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto p-0 gap-0 scrollbar-kasi">
   153→   153→        <DialogHeader className="px-6 pt-6 pb-0">
   154→   154→          <div className="flex items-center justify-between">
   155→   155→            <div className="flex items-center gap-3">
   156→   156→              <div className="relative w-10 h-10">
   157→   157→                <div className="absolute inset-0 rounded-xl bg-gradient-to-br from-emerald-500 to-amber-500" />
   158→   158→                <div className="absolute inset-0.5 rounded-[10px] bg-background flex items-center justify-center">
   159→   159→                  <span className="text-lg font-black bg-gradient-to-br from-emerald-600 to-amber-500 bg-clip-text text-transparent">K</span>
   160→   160→                </div>
   161→   161→              </div>
   162→   162→              <div>
   163→   163→                <DialogTitle className="text-xl">Join KaSiHUB</DialogTitle>
   164→   164→                <p className="text-xs text-muted-foreground">Become a member of the hybrid ecosystem</p>
   165→   165→              </div>
   166→   166→            </div>
   167→   167→            <Button variant="ghost" size="icon" onClick={closeRegistration} disabled={submitting}>
   168→   168→              <X className="h-4 w-4" />
   169→   169→            </Button>
   170→   170→          </div>
   171→   171→
   172→   172→          {/* Stepper */}
   173→   173→          {step !== "done" && (
   174→   174→            <div className="mt-6 flex items-center gap-2">
   175→   175→              {STEPS.map((s, i) => {
   176→   176→                const done = i < stepIndex;
   177→   177→                const active = i === stepIndex;
   178→   178→                return (
   179→   179→                  <div key={s.key} className="flex-1 flex items-center gap-2">
   180→   180→                    <div className={`flex items-center gap-2 ${active ? "text-emerald-600" : done ? "text-emerald-600" : "text-muted-foreground"}`}>
   181→   181→                      <div className={`w-7 h-7 rounded-full flex items-center justify-center text-xs font-bold border-2 transition-all ${
   182→   182→                        active ? "border-emerald-600 bg-emerald-600 text-white" :
   183→   183→                        done ? "border-emerald-600 bg-emerald-50 dark:bg-emerald-950 text-emerald-600" :
   184→   184→                        "border-border bg-background"
   185→   185→                      }`}>
   186→   186→                        {done ? <Check className="h-3.5 w-3.5" /> : i + 1}
   187→   187→                      </div>
   188→   188→                      <span className={`text-xs font-medium hidden sm:block ${active ? "" : "text-muted-foreground"}`}>{s.label}</span>
   189→   189→                    </div>
   190→   190→                    {i < STEPS.length - 1 && (
   191→   191→                      <div className={`flex-1 h-px ${done ? "bg-emerald-600" : "bg-border"}`} />
   192→   192→                    )}
   193→   193→                  </div>
   194→   194→                );
   195→   195→              })}
   196→   196→            </div>
   197→   197→          )}
   198→   198→        </DialogHeader>
   199→   199→
   200→   200→        <div className="p-6">
   201→   201→          <AnimatePresence mode="wait">
   202→   202→            <motion.div
   203→   203→              key={step}
   204→   204→              initial={{ opacity: 0, x: 20 }}
   205→   205→              animate={{ opacity: 1, x: 0 }}
   206→   206→              exit={{ opacity: 0, x: -20 }}
   207→   207→              transition={{ duration: 0.2 }}
   208→   208→            >
   209→   209→              {step === "type" && <TypeStep data={data} update={update} />}
   210→   210→              {step === "details" && <DetailsStep data={data} update={update} />}
   211→   211→              {step === "subscription" && <SubscriptionStep data={data} update={update} />}
   212→   212→              {step === "kyc" && <KycStep data={data} update={update} />}
   213→   213→              {step === "profile" && <ProfileStep data={data} update={update} />}
   214→   214→              {step === "done" && (
   215→   215→                <DoneStep profileNumber={newProfileNumber} membershipType={data.membershipType} onFinish={finish} />
   216→   216→              )}
   217→   217→            </motion.div>
   218→   218→          </AnimatePresence>
   219→   219→
   220→   220→          {step !== "done" && (
   221→   221→            <div className="mt-8 flex items-center justify-between">
   222→   222→              <Button variant="ghost" onClick={prev} disabled={stepIndex === 0 || submitting}>
   223→   223→                <ArrowLeft className="h-4 w-4 mr-2" /> Back
   224→   224→              </Button>
   225→   225→              {step === "profile" ? (
   226→   226→                <Button onClick={submit} disabled={submitting} className="bg-gradient-to-r from-emerald-600 to-emerald-500">
   227→   227→                  {submitting ? <><Loader2 className="h-4 w-4 mr-2 animate-spin" /> Creating profile...</> : <>Complete registration <Sparkles className="h-4 w-4 ml-2" /></>}
   228→   228→                </Button>
   229→   229→              ) : (
   230→   230→                <Button onClick={next} className="bg-gradient-to-r from-emerald-600 to-emerald-500">

   232→   232→                </Button>
   233→   233→              )}
   234→   234→            </div>
   235→   235→          )}
   236→   236→        </div>
   237→   237→      </DialogContent>
   238→   238→    </Dialog>
   239→   239→  );
   240→   240→}
