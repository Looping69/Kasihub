    46→    46→/** Derive a printable display name from a Member record. */
    47→    47→function memberDisplayName(member: Member | null): string {
    48→    48→  if (!member) return "—";
    49→    49→  const isCompany =
    50→    50→    member.membershipType === "COMPANY" ||
    51→    51→    member.membershipType === "NPO_NGO" ||
    52→    52→    member.membershipType === "SOLE_PROPRIETOR";
    53→    53→  if (isCompany && member.companyName) return member.companyName;
    54→    54→  const parts = [member.firstName, member.lastName].filter(Boolean);
    55→    55→  if (parts.length) return parts.join(" ");
    56→    56→  return member.email;
    57→    57→}
